1. Найди место где объявлен currentGuess:

var currentGuess by remember { mutableStateOf("") }

НИЖЕ добавь функцию:

fun rebuildGuessFromBoard() {
    val newGuess = CharArray(params.lenSecret) { '_' }

    for (digit in 0..9) {
        for (pos in 0 until params.lenSecret) {
            if (board[digit][pos] == CellMark.YES) {
                newGuess[pos] = ('0' + digit)
            }
        }
    }

    currentGuess = newGuess.map { if (it == '_') "" else it.toString() }.joinToString("")
}

2. Найди DigitsModule → onApplyToolToBoard

ЗАМЕНИ НА:

onApplyToolToBoard = { digit ->
    val mark = when (selectedTool) {
        TableTool.NO -> CellMark.NO
        TableTool.MAYBE -> CellMark.MAYBE
        TableTool.YES -> CellMark.YES
    }

    repeat(params.lenSecret) { position ->
        board[digit.digitToInt()][position] = mark
    }

    if (selectedTool == TableTool.YES) {
        rebuildGuessFromBoard()
    }
}
